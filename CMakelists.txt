cmake_minimum_required(VERSION 3.16)
project(iptvPlus VERSION 0.1.3 LANGUAGES CXX C)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE "Release" CACHE STRING "Choose Build Type" FORCE)
endif()

if(MSVC)
    add_compile_options(/Zc:__cplusplus)
endif()

set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)

set(THIRDPARTY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/3rdparty")

# Dependencias
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Svg Network sql)

set(PROJECT_SOURCES
    src/main.cpp
    src/mainwindow.cpp
    src/videoplayerwindow.cpp
    src/m3uparser.cpp
    src/channellistmodel.cpp
    src/movies.cpp
    include/movies.hpp
    src/dbmanager.cpp
    include/dbmanager.hpp
    include/channellistmodel.hpp
    include/mainwindow.hpp
    include/videoplayerwindow.hpp
    include/m3uparser.hpp
    include/update.hpp
    resources.qrc
    icon.rc
)

add_executable(${PROJECT_NAME} WIN32 ${PROJECT_SOURCES})


target_include_directories(${PROJECT_NAME} PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/include"
    "${THIRDPARTY_DIR}/mpv/include"
)

# Enlazar la librería explícitamente
target_link_libraries(${PROJECT_NAME} PRIVATE
    Qt6::Core
    Qt6::Gui
    Qt6::Widgets
    Qt6::Svg
    Qt6::Network
    Qt6::Sql
    #Qt6::EntryPointPrivate
    "${THIRDPARTY_DIR}/mpv/libmpv.dll.a"
)

include(GNUInstallDirs)

install(TARGETS ${PROJECT_NAME}
    RUNTIME DESTINATION .
)

# mpv
install(FILES
    "${THIRDPARTY_DIR}/mpv/libmpv-2.dll"
    DESTINATION .
)

# -----------------------------------------------------------------------------
# Compilar Script de Python a Ejecutable (PyInstaller)
# -----------------------------------------------------------------------------

find_program(PYINSTALLER_EXECUTABLE pyinstaller)

if(PYINSTALLER_EXECUTABLE)
    set(PYTHON_SCRIPT "${CMAKE_CURRENT_SOURCE_DIR}/src/iptvplus/main.py")
    set(PY_OUTPUT_DIR "${CMAKE_CURRENT_BINARY_DIR}/py_build")
    set(PY_EXE_NAME "scrap")
    set(PY_TARGET_EXE "${PY_OUTPUT_DIR}/dist/${PY_EXE_NAME}.exe")

    # Comando de compilación personalizada
    add_custom_command(
        OUTPUT "${PY_TARGET_EXE}"
        COMMAND "${PYINSTALLER_EXECUTABLE}"
                --noconfirm
                --onefile
                --name "${PY_EXE_NAME}"
                --exclude-module "urllib3.contrib.emscripten"
                --distpath "${PY_OUTPUT_DIR}/dist"
                --workpath "${PY_OUTPUT_DIR}/build"
                --specpath "${PY_OUTPUT_DIR}"
                "${PYTHON_SCRIPT}"
        DEPENDS "${PYTHON_SCRIPT}"
        COMMENT "Compilando script de Python con PyInstaller..."
        VERBATIM
    )

    # Crear objetivo dentro de MSBuild / Visual Studio
    add_custom_target(build_python_helper ALL
        DEPENDS "${PY_TARGET_EXE}"
    )

    add_dependencies(${PROJECT_NAME} build_python_helper)

    # Regla de instalación para CPack / NSIS
    install(FILES
        "${PY_TARGET_EXE}"
        DESTINATION .
    )
endif()

# -----------------------------------------------------------------------------
# Qt / windeployqt
# -----------------------------------------------------------------------------

find_program(WINDEPLOYQT_EXECUTABLE windeployqt)

if(WINDEPLOYQT_EXECUTABLE)

        execute_process(
            COMMAND \"${WINDEPLOYQT_EXECUTABLE}\"
