cmake_minimum_required(VERSION 4.0)
project(iptvPlus VERSION 0.2.1 LANGUAGES CXX)

# =============================================================================
# C++ CONFIGURATION
# =============================================================================

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Choose Build Type" FORCE)
endif()

if(MSVC)
    add_compile_options(/Zc:__cplusplus)
    #set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)

set(THIRDPARTY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/3rdparty")

# Forzar búsqueda de librerías estáticas para OpenSSL
set(OPENSSL_USE_STATIC_LIBS ON CACHE BOOL "" FORCE)

find_package(Qt6 REQUIRED COMPONENTS
    Core
    Gui
    Widgets
    Svg
    Network
    Sql
)

find_package(Boost REQUIRED COMPONENTS system)
find_package(OpenSSL REQUIRED)

if(APPLE)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(LIBTORRENT_RASTERBAR REQUIRED IMPORTED_TARGET libtorrent-rasterbar)
    add_library(LibtorrentRasterbar::torrent-rasterbar ALIAS PkgConfig::LIBTORRENT_RASTERBAR)
else()
    find_package(LibtorrentRasterbar REQUIRED)
endif()

# =============================================================================
# PROJECT SOURCES
# =============================================================================

set(PROJECT_SOURCES
    # Main
    src/main.cpp

    # Database
    src/databasemanager.cpp
    include/databasemanager.hpp

    # Torrent Engine
    src/torrentengine.cpp
    src/httpstreamserver.cpp
    include/httpstreamserver.hpp
    include/torrentengine.hpp

    # MPV
    src/mpvplayer.cpp
    include/mpvplayer.hpp

    # Video Player
    src/videoplayerwindow.cpp
    include/videoplayerwindow.hpp

    # Main Window
    src/mainwindow.cpp
    include/mainwindow.hpp

    # TV
    src/tv/channellistmodel.cpp
    src/tv/m3uparser.cpp
    src/tv/tv.cpp

    include/tv/channellistmodel.hpp
    include/tv/m3uparser.hpp
    include/tv/tv.hpp
    include/update.hpp

    # Movies
    src/movie/movies.cpp
    src/movie/moviedetail.cpp
    include/movie/moviedetail.hpp
    include/movie/movies.hpp

    # Resources
    resources.qrc
    icon.rc
)

# =============================================================================
# EXECUTABLE
# =============================================================================

add_executable(
    ${PROJECT_NAME}
    WIN32
    ${PROJECT_SOURCES}
)

# =============================================================================
# INCLUDE DIRECTORIES
# =============================================================================

target_include_directories(
    ${PROJECT_NAME}
    PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/include"
    "${CMAKE_CURRENT_SOURCE_DIR}/include/tv"
    "${CMAKE_CURRENT_SOURCE_DIR}/include/movie"
    "${THIRDPARTY_DIR}/mpv/include"
)

# =============================================================================
# COMPILE DEFINITIONS
# =============================================================================

target_compile_definitions(
    ${PROJECT_NAME}
    PRIVATE
    BOOST_ALL_NO_LIB
)

# =============================================================================
# LINK LIBRARIES
# =============================================================================

target_link_libraries(
    ${PROJECT_NAME}
    PRIVATE
    Qt6::Core
    Qt6::Gui
    Qt6::Widgets
    Qt6::Svg
    Qt6::Network
    Qt6::Sql
    Boost::system
    OpenSSL::SSL
    OpenSSL::Crypto
    LibtorrentRasterbar::torrent-rasterbar
)

if(APPLE)
    find_library(MPV_LIBRARY NAMES mpv REQUIRED)
    target_link_libraries(${PROJECT_NAME} PRIVATE ${MPV_LIBRARY})
else()
    target_link_libraries(${PROJECT_NAME} PRIVATE "${THIRDPARTY_DIR}/mpv/libmpv.dll.a")
endif()

# =============================================================================
# INSTALL
# =============================================================================

include(GNUInstallDirs)

# Ejecutable
install(
    TARGETS ${PROJECT_NAME}
    RUNTIME DESTINATION .
)

# MPV sigue siendo una DLL externa dinámica
install(
    FILES "${THIRDPARTY_DIR}/mpv/libmpv-2.dll"
    DESTINATION .
)

# =============================================================================
# PYTHON / PYINSTALLER
# =============================================================================

find_program(PYINSTALLER_EXECUTABLE pyinstaller)

if(PYINSTALLER_EXECUTABLE)
    set(PYTHON_SCRIPT "${CMAKE_CURRENT_SOURCE_DIR}/src/iptvplus/main.py")
    set(PY_OUTPUT_DIR "${CMAKE_CURRENT_BINARY_DIR}/py_build")
    set(PY_EXE_NAME scrap)
    
    if(WIN32)
        set(PY_TARGET_EXE "${PY_OUTPUT_DIR}/dist/${PY_EXE_NAME}.exe")
    else()
        set(PY_TARGET_EXE "${PY_OUTPUT_DIR}/dist/${PY_EXE_NAME}")
    endif()

    file(GLOB_RECURSE ALL_PYTHON_SCRIPTS "${CMAKE_CURRENT_SOURCE_DIR}/src/iptvplus/*.py")

    add_custom_target(
        build_python_helper ALL
        COMMAND "${PYINSTALLER_EXECUTABLE}"
            --noconfirm
            --onefile
            --name "${PY_EXE_NAME}"
            --paths "${CMAKE_CURRENT_SOURCE_DIR}/src/iptvplus"
            --exclude-module urllib3.contrib.emscripten
            --distpath "${PY_OUTPUT_DIR}/dist"
            --workpath "${PY_OUTPUT_DIR}/build"
            --specpath "${PY_OUTPUT_DIR}"
            "${PYTHON_SCRIPT}"
        BYPRODUCTS "${PY_TARGET_EXE}"
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
        COMMENT "Compilando scraper Python con PyInstaller..."
        VERBATIM
    )

    add_dependencies(${PROJECT_NAME} build_python_helper)

    install(
        FILES "${PY_TARGET_EXE}"
        DESTINATION .
    )
endif()

# =============================================================================
# QT DEPLOYMENT
# =============================================================================

find_program(WINDEPLOYQT_EXECUTABLE windeployqt)

        "
    )
endif()
