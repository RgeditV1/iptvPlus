cmake_minimum_required(VERSION 3.16)
project(iptvPlus VERSION 0.1.1 LANGUAGES CXX C)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE "Release" CACHE STRING "Choose Build Type" FORCE)
endif()

if(MSVC)
    add_compile_options(/Zc:__cplusplus)
endif()

set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)

set(THIRDPARTY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/3rdparty")

# Dependencias
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Svg Network)

set(PROJECT_SOURCES
    src/main.cpp
    src/mainwindow.cpp
    src/videoplayerwindow.cpp
    src/m3uparser.cpp
    src/channellistmodel.cpp
    include/channellistmodel.hpp
    include/mainwindow.hpp
    include/videoplayerwindow.hpp
    include/m3uparser.hpp
    include/update.hpp
    resources.qrc
)

add_executable(${PROJECT_NAME} WIN32 ${PROJECT_SOURCES})


target_include_directories(${PROJECT_NAME} PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/include"
    "${THIRDPARTY_DIR}/mpv/include"
)

# Enlazar la librería explícitamente
target_link_libraries(${PROJECT_NAME} PRIVATE
    Qt6::Core
    Qt6::Gui
    Qt6::Widgets
    Qt6::Svg
    Qt6::Network
    #Qt6::EntryPointPrivate
    "${THIRDPARTY_DIR}/mpv/libmpv.dll.a"
)

include(GNUInstallDirs)

install(TARGETS ${PROJECT_NAME}
    RUNTIME DESTINATION .
)

# mpv
install(FILES
    "${THIRDPARTY_DIR}/mpv/libmpv-2.dll"
    DESTINATION .
)

# -----------------------------------------------------------------------------
# Qt / windeployqt
# -----------------------------------------------------------------------------

find_program(WINDEPLOYQT_EXECUTABLE windeployqt)

if(WINDEPLOYQT_EXECUTABLE)
    install(CODE "
        execute_process(
            COMMAND \"${WINDEPLOYQT_EXECUTABLE}\"
                    --release
                    --no-translations
                    \"\${CMAKE_INSTALL_PREFIX}/${PROJECT_NAME}.exe\"
        )
    ")
endif()

# -----------------------------------------------------------------------------
# REDIST
# -----------------------------------------------------------------------------

set(REDIST_DIR "${CMAKE_CURRENT_SOURCE_DIR}/_redist")

install(FILES
    "${REDIST_DIR}/vcredist_v14.x64.exe"
    "${REDIST_DIR}/vcredist_v14.x86.exe"
    "${REDIST_DIR}/VulkanRT-Installer.exe"
    DESTINATION "_redist"
)

# -----------------------------------------------------------------------------
# CPACK
# -----------------------------------------------------------------------------

set(CPACK_PACKAGE_NAME "iptvPlus")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")

set(CPACK_GENERATOR "NSIS;ZIP")

set(CPACK_NSIS_EXTRA_INSTALL_COMMANDS "
    ExecWait '\\\"\\$INSTDIR\\\\_redist\\\\vcredist_v14.x64.exe\\\" /quiet /norestart'
    ExecWait '\\\"\\$INSTDIR\\\\_redist\\\\vcredist_v14.x86.exe\\\" /quiet /norestart'
    ExecWait '\\\"\\$INSTDIR\\\\_redist\\\\VulkanRT-Installer.exe\\\" /S'
    RMDir /r '\\$INSTDIR\\\\_redist'
")

include(CPack)
